  // Worker orchestration logic
  async function fetchJob(workerId: string) {
    return db.transaction(async tx => {
      const job = await tx.query(`
        UPDATE jobs SET 
          status = 'processing', 
          worker_id = $1, 
          started_at = NOW()
        WHERE id = (
          SELECT id FROM jobs 
          WHERE status = 'queued' AND scheduled_at <= NOW()
          FOR UPDATE SKIP LOCKED
          LIMIT 1
        ) RETURNING *
      `, [workerId]);
      return job.rows[0];
    });
